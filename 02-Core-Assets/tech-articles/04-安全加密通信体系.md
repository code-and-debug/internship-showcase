# 医疗系统安全加密通信体系设计

## 一、安全需求分析

### 1.1 医疗数据的特殊性

医疗数据属于**敏感个人信息**，需要满足：
- **等保2.0三级要求**
- **《个人信息保护法》合规**
- **《数据安全法》合规**
- **防止中间人攻击**
- **防止重放攻击**

### 1.2 攻击面分析

| 攻击类型 | 风险等级 | 防护措施 |
|---------|---------|---------|
| 中间人窃听 | 高 | HTTPS + 请求体加密 |
| 重放攻击 | 高 | 时间戳签名 |
| Token盗用 | 中 | Token过期机制 |
| 暴力破解 | 中 | 限流 + 验证码 |
| 逆向工程 | 中 | 代码混淆 |

## 三、核心实现

### 3.1 加密工具类

```javascript
// utils/des.js
import CryptoJS from 'crypto-js';

const MY_KEY = 'W7ZEgfnv'; // 默认密钥

/**
 * DES加密
 * @param {string} message - 待加密文本
 * @param {string} key - 密钥，默认MY_KEY
 * @returns {string} - Base64加密结果
 */
export const encryptDes = (message, key = MY_KEY) => {
  const keyHex = CryptoJS.enc.Utf8.parse(key);
  const encrypted = CryptoJS.DES.encrypt(message, keyHex, {
    mode: CryptoJS.mode.ECB,
    padding: CryptoJS.pad.Pkcs7,
  });
  return encrypted.toString();
};

/**
 * DES解密
 * @param {string} ciphertext - Base64密文
 * @param {string} key - 密钥
 * @returns {string} - 解密后的明文
 */
export const decryptDes = (ciphertext, key = MY_KEY) => {
  const keyHex = CryptoJS.enc.Utf8.parse(key);
  const decrypted = CryptoJS.DES.decrypt(
    {
      ciphertext: CryptoJS.enc.Base64.parse(ciphertext),
    },
    keyHex,
    {
      mode: CryptoJS.mode.ECB,
      padding: CryptoJS.pad.Pkcs7,
    }
  );
  return decrypted.toString(CryptoJS.enc.Utf8);
};

/**
 * AES加密
 * @param {string} val - 待加密文本
 * @param {string} key - 密钥
 * @returns {string}
 */
export const encryptedAes = (val, key) => {
  key = CryptoJS.enc.Utf8.parse(key);
  return CryptoJS.AES.encrypt(val, key, {
    mode: CryptoJS.mode.ECB,
    padding: CryptoJS.pad.Pkcs7,
  }).toString();
};

/**
 * JSON对象加密（先URI编码再AES）
 * @param {Object} iJson - JSON对象
 * @param {string} key - 密钥
 * @returns {string}
 */
export const useEncryptAes = (iJson, key = MY_KEY) => {
  return encryptedAes(encodeURIComponent(JSON.stringify(iJson)), key);
};
```

### 3.2 请求拦截器（核心）

```javascript
// utils/request.js
import { extend } from 'umi-request';
import { notification } from 'antd';
import { getDvaApp } from 'umi';
import { encryptDes, decryptDes } from '@/utils/des';
import globleData from '@/utils/globleData';

// 错误码映射
const codeMessage = {
  200: '服务器成功返回请求的数据',
  401: '用户没有权限（令牌、用户名、密码错误）',
  403: '用户得到授权，但是访问是被禁止的',
  500: '服务器发生错误',
  502: '网关错误',
  503: '服务不可用',
};

// 统一错误处理
const errorHandler = (error) => {
  const { response } = error;
  if (response && response.status) {
    const errorText = codeMessage[response.status] || response.statusText;
    notification.error({
      message: `请求错误 ${response.status}`,
      description: errorText,
    });
  } else if (!response) {
    notification.error({
      description: '您的网络发生异常，无法连接服务器',
      message: '网络异常',
    });
  }
  return response;
};

// 创建请求实例
const request = extend({
  prefix: REACT_APP_ENV === 'development' 
    ? 'https://testphs.eheren.com/gateway' 
    : 'https://netphs.eheren.com/gateway',
  errorHandler,
  timeout: 10000, // 10秒超时
});

// ==================== 请求拦截器 ====================
request.interceptors.request.use((url, options) => {
  // 1. 生成时间戳签名（防重放攻击）
  // 格式: sysCode + '_' + 时间戳
  const timestamp = new Date().getTime();
  const signContent = `${globleData.sysCode}_${timestamp}`;
  const sysCodeSign = encryptDes(signContent, 'W7ZEgfnv');
  
  // 2. 设置请求头
  options.headers = {
    ...options.headers,
    phsId: '81681688',      // 固定标识
    phsSign: sysCodeSign,   // 动态签名
    hrCode: encryptDes(globleData.sysCode, 'hrtest22'),
  };
  
  // 3. 注入用户标识
  options.data = options.data || {};
  options.data.psnId = localStorage.getItem('herenId');
  options.data.herenId = localStorage.getItem('herenId');
  
  // 4. 包裹参数结构
  if (!options.noNeedWrapArgs) {
    options.data = {
      args: options.data,
      token: localStorage.getItem('token'),
    };
  }
  
  // 5. 生产环境加密（核心安全措施）
  if (REACT_APP_ENV === 'production') {
    // 动态密钥：每天更换
    // reqv3-01, reqv3-02, ..., reqv3-31
    const day = ('0' + new Date().getDate()).slice(-2);
    const key = `reqv3-${day}`;
    
    const data = JSON.parse(JSON.stringify(options.data));
    const desData = {
      args: {},
      signContent: encryptDes(JSON.stringify(data.args), key),
      token: data.token,
    };
    options.data = desData;
    
    // 网关限流标识
    url = url + '=' + encryptDes(globleData.sysCode, 'hrtest22');
  }
  
  return {
    url,
    options: { ...options, interceptors: true },
  };
});

// ==================== 响应拦截器 ====================
request.interceptors.response.use(async (response, options) => {
  // 克隆响应（因为response只能读取一次）
  const data = await response.clone().json();
  
  // 1. 生产环境解密
  if (REACT_APP_ENV === 'production' && data.signContent) {
    const day = ('0' + new Date().getDate()).slice(-2);
    const key = `resv3-${day}`;
    try {
      data.result = JSON.parse(decryptDes(data.signContent, key));
    } catch (e) {
      console.error('解密失败', e);
      notification.error({ message: '数据解密失败' });
    }
  }
  
  // 2. Token过期处理（统一认证中心返回码）
  // 4000: Token不存在  4001: Token过期  4002: Token无效  4003: Token被踢出
  if (['4000', '4001', '4002', '4003'].includes(data.code)) {
    notification.error({
      message: '登录过期，请重新登录',
    });
    
    // 延迟2秒后登出，让用户看到提示
    setTimeout(() => {
      getDvaApp()._store.dispatch({
        type: 'login/logout',
      });
    }, 2000);
  }
  
  // 3. 业务错误统一提示
  if (data.code !== '0' && data.message) {
    notification.error({
      message: data.message,
    });
  }
  
  return data;
});

export default request;
```

## 四、安全机制详解

### 4.1 防重放攻击

```javascript
// 原理：时间戳签名，服务端校验时间差

// 前端生成签名
const timestamp = new Date().getTime();
const sign = encryptDes(`${sysCode}_${timestamp}`, key);

// 后端校验逻辑（伪代码）
function verifySign(sign) {
  const [sysCode, timestamp] = decrypt(sign).split('_');
  const now = Date.now();
  const diff = now - parseInt(timestamp);
  
  // 时间差不能超过5分钟
  if (diff > 5 * 60 * 1000) {
    throw new Error('请求已过期，请重试');
  }
  
  // 检查是否已使用过（Redis去重）
  if (redis.get(sign)) {
    throw new Error('重复请求');
  }
  redis.setex(sign, 300, '1'); // 5分钟过期
}
```

### 4.2 动态密钥机制

```javascript
// 每日自动更换密钥，降低密钥泄露风险
const getDailyKey = (prefix) => {
  const day = ('0' + new Date().getDate()).slice(-2);
  return `${prefix}${day}`;
};

// 请求密钥: reqv3-01 ~ reqv3-31
const requestKey = getDailyKey('reqv3-');

// 响应密钥: resv3-01 ~ resv3-31  
const responseKey = getDailyKey('resv3-');

// 1. 密钥由后端生成，通过安全通道下发
```

### 4.3 Token安全策略

```javascript
// models/login.js
const loginModel = {
  effects: {
    *login({ payload }, { call, put }) {
      const res = yield call(fakeAccountLogin, payload);
      if (res.code === '0') {
        const { accessToken, refreshToken, expiresIn } = res.result;
        
        // 存储Token
        localStorage.setItem('token', accessToken);
        localStorage.setItem('refreshToken', refreshToken);
        
        // 计算过期时间
        const expireTime = Date.now() + expiresIn * 1000;
        localStorage.setItem('tokenExpire', expireTime);
      }
    },
    
    // Token刷新
    *refreshToken(_, { call, put }) {
      const refreshToken = localStorage.getItem('refreshToken');
      const res = yield call(refreshTokenApi, { refreshToken });
      
      if (res.code === '0') {
        localStorage.setItem('token', res.result.accessToken);
      } else {
        // 刷新失败，重新登录
        yield put({ type: 'logout' });
      }
    },
  },
};

// 请求前检查Token是否即将过期
const checkTokenExpire = () => {
  const expireTime = localStorage.getItem('tokenExpire');
  const now = Date.now();
  
  // 提前5分钟刷新
  if (expireTime - now < 5 * 60 * 1000) {
    store.dispatch({ type: 'login/refreshToken' });
  }
};
```

## 七、面试话术

**Q: 你们系统是怎么保证数据安全的？**

> "我们从传输层和应用层两个维度保障安全：
> 
> **传输层**：
> - 全站HTTPS，启用HSTS
> - 请求体DES加密，生产环境自动启用
> 
> **应用层**：
> - **防重放攻击**：每个请求带时间戳签名，服务端校验5分钟有效期
> - **动态密钥**：密钥每天更换（reqv3-01到reqv3-31）
> - **Token安全**：双Token机制，Access Token 2小时过期，Refresh Token 7天过期
> - **自动续期**：请求拦截器检查Token即将过期时自动刷新
> 
> **架构层面**：
> - 统一错误处理，不暴露内部错误细节
> - 敏感字段（身份证、手机号）日志脱敏"

**Q: 如果密钥泄露了怎么办？**

> "我们有多层防护：
> 1. **动态密钥**：每天自动更换，泄露影响范围小
> 2. **签名机制**：即使知道密钥，没有正确签名也无法通过校验
> 3. **时间戳校验**：重放旧请求会被拒绝
> 4. **应急响应**：一旦泄露，立即推送新版本更换密钥体系"

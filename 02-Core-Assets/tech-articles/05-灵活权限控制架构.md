# 灵活权限控制架构设计 - RBAC策略模式实现

## 一、需求分析

### 1.1 权限场景

医院管理系统中，权限控制涉及多个层面：

| 层面 | 示例 | 技术方案 |
|-----|------|---------|
| 路由级 | 管理员才能访问后台 | 路由守卫 |
| 菜单级 | 某些角色看不到某菜单 | 动态菜单渲染 |
| 组件级 | 普通用户看不到删除按钮 | 权限组件包裹 |
| 按钮级 | 仅科主任可审批 | 指令/高阶组件 |
| 数据级 | 医生只能看自己的患者 | 后端数据过滤 |

### 1.2 传统方案的痛点

```javascript
// ❌ 方案1: 硬编码if-else
if (user.role === 'admin') {
  return <AdminPage />;
} else if (user.role === 'doctor') {
  return <DoctorPage />;
}
// 问题: 角色多了维护困难

// ❌ 方案2: 简单的数组includes
const canAccess = ['admin', 'doctor'].includes(user.role);
// 问题: 不支持复杂逻辑（如：admin且vip）

// ❌ 方案3: 前端控制路由
if (!isAdmin) {
  history.push('/403');
}
// 问题: 用户体验差，会闪烁
```

## 二、架构设计

### 2.1 核心思想：策略模式

将权限判定逻辑抽象为**策略**，支持多种判定方式：

```javascript
// 策略类型
const AUTH_STRATEGY = {
  STRING: 'string',       // 字符串匹配: 'admin'
  ARRAY: 'array',         // 数组匹配: ['admin', 'doctor']
  FUNCTION: 'function',   // 函数判定: (role) => role.includes('admin')
  PROMISE: 'promise',     // 异步判定: Promise.resolve(true)
};
```

## 三、核心实现

### 3.1 权限检查引擎

```javascript
// components/Authorized/CheckPermissions.jsx
import React from 'react';
import PromiseRender from './PromiseRender';
import { CURRENT } from './renderAuthorize';

/**
 * 通用权限检查方法 - 策略模式实现
 * @param {String|Array|Function|Promise} authority - 权限判定
 * @param {String|Array} currentAuthority - 当前用户权限
 * @param {ReactNode} target - 通过的组件
 * @param {ReactNode} Exception - 未通过的组件
 */
const checkPermissions = (authority, currentAuthority, target, Exception) => {
  // ========== 策略1: 无权限要求，默认通过 ==========
  if (!authority) {
    return target;
  }
  
  // ========== 策略2: 数组权限匹配 ==========
  if (Array.isArray(authority)) {
    if (Array.isArray(currentAuthority)) {
      // 多对多: 用户有任一权限即可
      if (currentAuthority.some((item) => authority.includes(item))) {
        return target;
      }
    } else if (authority.includes(currentAuthority)) {
      // 多对一
      return target;
    }
    return Exception;
  }
  
  // ========== 策略3: 字符串权限匹配 ==========
  if (typeof authority === 'string') {
    if (Array.isArray(currentAuthority)) {
      // 一对多
      if (currentAuthority.some((item) => authority === item)) {
        return target;
      }
    } else if (authority === currentAuthority) {
      // 一对一
      return target;
    }
    return Exception;
  }
  
  // ========== 策略4: Promise异步权限 ==========
  if (authority instanceof Promise) {
    return (
      <PromiseRender
        ok={target}
        error={Exception}
        promise={authority}
      />
    );
  }
  
  // ========== 策略5: 函数自定义权限 ==========
  if (typeof authority === 'function') {
    try {
      const bool = authority(currentAuthority);
      
      // 函数返回Promise
      if (bool instanceof Promise) {
        return (
          <PromiseRender
            ok={target}
            error={Exception}
            promise={bool}
          />
        );
      }
      
      // 函数返回布尔值
      if (bool) {
        return target;
      }
      return Exception;
    } catch (error) {
      throw error;
    }
  }
  
  throw new Error('unsupported parameters');
};

// 简化调用方式（使用CURRENT作为当前权限）
export const check = (authority, target, Exception) => {
  return checkPermissions(authority, CURRENT, target, Exception);
};

export default checkPermissions;
```

### 3.2 Promise权限渲染器

```javascript
// components/Authorized/PromiseRender.jsx
import React from 'react';
import { Spin } from 'antd';

/**
 * 异步权限渲染组件
 * 用于处理Promise类型的权限判定
 */
class PromiseRender extends React.PureComponent {
  state = {
    component: null, // 当前要渲染的组件
  };
  
  componentDidMount() {
    this.setRenderComponent(this.props);
  }
  
  componentDidUpdate(nextProps) {
    // promise变化时重新判定
    if (nextProps.promise !== this.props.promise) {
      this.setRenderComponent(nextProps);
    }
  }
  
  // 异步判定权限
  setRenderComponent(props) {
    const ok = this.checkIsInstantiation(props.ok);
    const error = this.checkIsInstantiation(props.error);
    
    props.promise
      .then(() => {
        this.setState({ component: ok });
      })
      .catch(() => {
        this.setState({ component: error });
      });
  }
  
  // 检查组件是否已实例化
  checkIsInstantiation = (target) => {
    if (!React.isValidElement(target)) {
      return target;
    }
    return () => target;
  };
  
  render() {
    const Component = this.state.component;
    
    // 加载中状态
    if (!Component) {
      return (
        <div style={{ padding: '20px', textAlign: 'center' }}>
          <Spin size="large" tip="权限校验中..." />
        </div>
      );
    }
    
    return <Component {...this.props} />;
  }
}

export default PromiseRender;
```

### 3.3 Authorized组件

```javascript
// components/Authorized/Authorized.jsx
import React from 'react';
import { Result } from 'antd';
import check from './CheckPermissions';

/**
 * 权限控制组件
 * 
 * 用法:
 * <Authorized authority="admin" noMatch={<NoPermission />}>
 *   <AdminContent />
 * </Authorized>
 */
const Authorized = ({
  children,
  authority,
  noMatch = (
    <Result
      status="403"
      title="403"
      subTitle="Sorry, you are not authorized to access this page."
    />
  ),
}) => {
  const childrenRender = typeof children === 'undefined' ? null : children;
  return <>{check(authority, childrenRender, noMatch)}</>;
};

export default Authorized;
```

### 3.4 权限渲染器

```javascript
// components/Authorized/renderAuthorize.js
/**
 * 权限渲染器
 * 用于动态更新CURRENT权限
 */
let CURRENT = 'NULL';

const renderAuthorize = (Authorized) => (currentAuthority) => {
  if (currentAuthority) {
    if (typeof currentAuthority === 'function') {
      CURRENT = currentAuthority();
    } else if (
      typeof currentAuthority === 'string' ||
      Array.isArray(currentAuthority)
    ) {
      CURRENT = currentAuthority;
    }
  } else {
    CURRENT = 'NULL';
  }
  
  return Authorized;
};

export { CURRENT };
export default (Authorized) => renderAuthorize(Authorized);
```

### 3.6 权限存储

```javascript
// utils/authority.js
import { reloadAuthorized } from './Authorized';

/**
 * 从localStorage获取权限
 */
export function getAuthority(str) {
  const authorityString =
    typeof str === 'undefined' && localStorage
      ? localStorage.getItem('antd-pro-authority')
      : str;
  
  let authority;
  try {
    if (authorityString) {
      authority = JSON.parse(authorityString);
    }
  } catch (e) {
    authority = authorityString;
  }
  
  if (typeof authority === 'string') {
    return [authority];
  }
  
  // 开发环境默认权限
  if (!authority && process.env.NODE_ENV === 'development') {
    return ['admin'];
  }
  
  return authority;
}

/**
 * 设置权限并重新加载
 */
export function setAuthority(authority) {
  const proAuthority = typeof authority === 'string' ? [authority] : authority;
  localStorage.setItem('antd-pro-authority', JSON.stringify(proAuthority));
  
  // 自动触发权限更新
  reloadAuthorized();
}
```

### 4.1 路由级权限

```javascript
// pages/Authorized.jsx
import React from 'react';
import { Redirect, connect } from 'umi';
import Authorized from '@/utils/Authorized';
import { getRouteAuthority } from '@/utils/utils';

const AuthComponent = ({
  children,
  route = { routes: [] },
  location = { pathname: '' },
  user,
}) => {
  const { currentUser } = user;
  const { routes = [] } = route;
  const isLogin = currentUser && currentUser.userid;
  
  // 获取当前路由需要的权限
  const authority = getRouteAuthority(location.pathname, routes) || '';
  
  return (
    <Authorized
      authority={authority}
      noMatch={
        isLogin ? (
          <Redirect to="/exception/403" />
        ) : (
          <Redirect to="/user/login" />
        )
      }
    >
      {children}
    </Authorized>
  );
};

export default connect(({ user }) => ({ user }))(AuthComponent);
```

```javascript
// 场景：医生只能编辑自己的患者，科主任可编辑所有
const canEditPatient = (currentAuthority, patient) => {
  // 管理员总是可以
  if (currentAuthority.includes('admin')) {
    return true;
  }
  
  // 科主任可以编辑科室所有患者
  if (currentAuthority.includes('dept_chief')) {

  // 普通医生只能编辑自己的患者

    return patient.doctorId === currentUser.userId;
  }
  
  return false;
};
```

### 5.2 权限Hooks

```javascript
// hooks/useAuthority.js
import { useState, useEffect } from 'react';
import { getAuthority } from '@/utils/authority';

export default function useAuthority(expectedAuthority) {
  const [hasAuthority, setHasAuthority] = useState(false);
  
  useEffect(() => {
    const current = getAuthority();
    
    if (Array.isArray(expectedAuthority)) {
      setHasAuthority(
        expectedAuthority.some((auth) => current.includes(auth))
      );
    } else {
      setHasAuthority(current.includes(expectedAuthority));
    }
  }, [expectedAuthority]);
  
  return hasAuthority;
}

// 使用
const canEdit = useAuthority(['admin', 'editor']);
```

## 七、面试话术

**Q: 你们权限控制是怎么实现的？**

> "我们采用的是**策略模式**实现的权限控制架构，支持五种权限判定方式：
> 
> 1. **字符串匹配**：简单角色如'admin'
> 2. **数组匹配**：多角色如['admin', 'doctor']
> 3. **函数判定**：复杂逻辑如(role) => role.includes('admin') && isVip
> 4. **Promise异步**：从服务器获取权限状态
> 5. **无权限**：默认通过
> 
> **应用层面**支持四级控制：
> - 路由级：未登录重定向到登录页
> - 菜单级：无权限的菜单自动隐藏
> - 组件级：用Authorized组件包裹，无权限显示空或403
> - 按钮级：自定义AuthButton组件
> 
> 比如医生只能编辑自己的患者，科主任可编辑全科室，我们用函数权限实现：
> ```javascript
> authority={(role) => {
>   if (role.includes('admin')) return true;
>   if (role.includes('dept_chief')) return patient.deptId === currentDeptId;
>   return patient.doctorId === currentUserId;
> }}
> ```"

**Q: 权限和路由怎么结合？**

> "我们在Umi路由配置中添加authority字段，在路由守卫组件中：
> 1. 用getRouteAuthority获取当前路由所需权限
> 2. 用Authorized组件包裹，根据权限决定渲染还是跳403
> 3. 菜单用Authorized.check动态过滤
> 
> 这样路由、菜单、页面组件的权限保持一致。"

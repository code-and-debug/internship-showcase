interface PatientRegistrationFlow {
  // 步骤1: 查询HIS是否存在档案
  async queryExistingPatient(idCard: string): Promise<PatientQueryResult> {
    const result = await hisApi.queryPatient({ idCard });
    
    if (result.exists) {
      // 已存在 → 进入绑定流程
      return this.enterBindingFlow(result.patientInfo);
    }
    // 不存在 → 进入建档流程
    return this.enterCreationFlow();
  }

  // 步骤2: 建档流程
  async createPatient(info: PatientInfo): Promise<Patient> {
    // 14岁以下强制监护人信息
    if (this.isMinor(info.birthDate)) {
      this.validateGuardianInfo(info.guardian);
    }
    
    // 证件格式校验
    this.validateIdCardFormat(info.idCardType, info.idCard);
    
    // 调用HIS建档接口
    const patient = await hisApi.createPatient(info);
    return patient;
  }

  // 步骤3: 绑定流程（含人脸核身判断）
  async bindPatient(userId: string, patientInfo: PatientInfo): Promise<BindingResult> {
    // 信息一致性校验
    const consistency = await this.checkInfoConsistency(patientInfo);
    
    if (!consistency.phoneMatch) {
      // 手机号不一致 → 触发人脸核身
      return this.triggerFaceVerification(patientInfo);
    }
    
    // 直接绑定
    return this.doBinding(userId, patientInfo.patientId);
  }
}

### 3.2 挂号预约域：号源管理与锁号机制

#### 号源库存模型

**核心业务概念：**
- **排班（Schedule）**：医生某天的出班计划
- **号源（Source）**：具体到时段的可预约名额
- **锁号（Lock）**：15分钟预占机制（防止并发超卖）

**号源状态流转：**

```
初始状态: 可用
  ↓ 用户点击预约
锁号中（15分钟）← Redis分布式锁
  ↓ 支付成功
已预约（确认号源）
  ↓ 就诊前1小时
待就诊（可在线取号）
  ↓ 患者取号/医生接诊
就诊中 → 已就诊

异常流：
- 锁号中 + 15分钟超时 → 自动释放（回滚库存）
- 锁号中 + 主动取消 → 立即释放
- 已预约 + 医生停诊 → 强制取消+退款
- 已预约 + 患者爽约 → 记违约（影响后续预约权限）
```

#### 锁号状态机

```typescript
// 伪代码：预约状态机
enum AppointmentStatus {
  LOCKED = 'locked',           // 锁号中（15分钟）
  PENDING_PAYMENT = 'pending', // 待支付
  CONFIRMED = 'confirmed',     // 已预约
  WAITING = 'waiting',         // 待就诊
  IN_PROGRESS = 'in_progress', // 就诊中
  COMPLETED = 'completed',     // 已就诊
  CANCELLED = 'cancelled',     // 已取消
  EXPIRED = 'expired',         // 已过期
  STOPPED = 'stopped'          // 已停诊
}

class AppointmentStateMachine {
  // 状态流转规则
  private transitions: Record<AppointmentStatus, AppointmentStatus[]> = {
    [AppointmentStatus.LOCKED]: [
      AppointmentStatus.PENDING_PAYMENT,
      AppointmentStatus.EXPIRED
    ],
    [AppointmentStatus.PENDING_PAYMENT]: [
      AppointmentStatus.CONFIRMED,
      AppointmentStatus.CANCELLED
    ],
    [AppointmentStatus.CONFIRMED]: [
      AppointmentStatus.WAITING,
      AppointmentStatus.CANCELLED,
      AppointmentStatus.STOPPED
    ],
    [AppointmentStatus.WAITING]: [
      AppointmentStatus.IN_PROGRESS,
      AppointmentStatus.EXPIRED
    ],
    [AppointmentStatus.IN_PROGRESS]: [AppointmentStatus.COMPLETED],
    // ... 其他状态
  };

  async transition(
    currentStatus: AppointmentStatus, 
    targetStatus: AppointmentStatus,
    context: TransitionContext
  ): Promise<boolean> {
    // 验证流转是否合法
    if (!this.transitions[currentStatus]?.includes(targetStatus)) {
      throw new InvalidTransitionError(`${currentStatus} -> ${targetStatus}`);
    }
    
    // 执行状态流转
    return this.executeTransition(targetStatus, context);
  }
}
```

#### 医保结算流程

**产品规则：**
1. **预结算**：支付前必须先调用医保接口进行费用分割，明确"统筹支付"与"个人自付"金额
2. **额度锁定**：医保部分需在医保平台锁定额度，防止支付过程中额度不足
3. **对账机制**：T+1日对账，差异订单自动冲正

```typescript
// 伪代码：医保结算流程
interface MedicalInsuranceSettlement {
  // 步骤1: 预结算 - 费用分割
  async preSettlement(orderId: string): Promise<PreSettlementResult> {
    const order = await this.getOrder(orderId);
    
    // 调用医保接口进行费用分割
    const splitResult = await medicalInsuranceApi.calculateSplit({
      patientId: order.patientId,
      totalAmount: order.totalAmount,
      items: order.items
    });
    
    return {
      totalAmount: order.totalAmount,
      insuranceCoverage: splitResult.coverage,    // 统筹支付
      personalPayment: splitResult.personal,      // 个人自付
      available: splitResult.insuranceAvailable   // 医保额度是否充足
    };
  }

  // 步骤2: 支付调起（自费部分）
  async initiatePayment(preSettlement: PreSettlementResult): Promise<PaymentIntent> {
    if (!preSettlement.available) {
      throw new InsuranceQuotaInsufficientError();
    }
    
    // 锁定医保额度
    await medicalInsuranceApi.lockQuota(preSettlement.insuranceCoverage);
    
    // 调起支付（仅个人自付部分）
    return paymentGateway.createPayment({
      amount: preSettlement.personalPayment,
      orderId: preSettlement.orderId
    });
  }

  // 步骤3: 支付回调确认
  async confirmSettlement(
    orderId: string, 
    paymentResult: PaymentResult
  ): Promise<SettlementResult> {
    if (paymentResult.success) {
      // 确认医保结算
      await medicalInsuranceApi.confirmSettlement(orderId);
      // 确认HIS结算
      await hisApi.confirmPayment(orderId);
      
      return { status: 'success', receiptNo: generateReceiptNo() };
    } else {
      // 释放医保额度
      await medicalInsuranceApi.releaseQuota(orderId);
      throw new PaymentFailedError();
    }
  }
}

```typescript
// 伪代码：配送策略路由引擎

// 策略配置定义
interface DeliveryStrategy {
  id: string;
  allowedMethods: DeliveryMethod[];
  defaultMethod: DeliveryMethod;
  requireAddress: boolean;
  validate: (context: DeliveryContext) => ValidationResult;
}

// 策略配置表（12种组合）
const DELIVERY_STRATEGIES: Record<string, DeliveryStrategy> = {
  // 标准流程 + 自煎药
  'standard-self': {
    id: 'standard-self',
    allowedMethods: ['window', 'address_only'],
    defaultMethod: 'window',
    requireAddress: (method) => method === 'address_only',
    validate: (ctx) => ({ valid: true })
  },
  
  // 标准流程 + 代煎药
  'standard-proxy': {
    id: 'standard-proxy',
    allowedMethods: ['window', 'express', 'address_only'],
    defaultMethod: 'window',
    requireAddress: (method) => ['express', 'address_only'].includes(method),
    validate: (ctx) => {
      // 代煎药不能仅填地址（必须实际交付）
      if (ctx.selectedMethod === 'address_only') {
        return { valid: false, message: '代煎药品请选择窗口自取或快递配送' };
      }
      return { valid: true };
    }
  },
  
  // 扫码直达 + 自煎药
  'scan-self': {
    id: 'scan-self',
    allowedMethods: ['window'],
    defaultMethod: 'window',
    requireAddress: false,
    validate: (ctx) => ({ valid: true })
  },
  
  // 扫码直达 + 代煎药
  'scan-proxy': {
    id: 'scan-proxy',
    allowedMethods: ['window', 'express'],
    defaultMethod: 'window',
    requireAddress: (method) => method === 'express',
    validate: (ctx) => ({ valid: true })
  }
};

// 策略路由引擎
class DeliveryStrategyRouter {
  resolveStrategy(context: DeliveryContext): DeliveryStrategy {
    // 根据入口类型和煎药类型匹配策略
    const key = `${context.entryType}-${context.decoctionType}`;
    const strategy = DELIVERY_STRATEGIES[key];
    
    if (!strategy) {
      throw new StrategyNotFoundError(key);
    }
    
    return strategy;
  }
  
  getAvailableOptions(context: DeliveryContext): DeliveryOptions {
    const strategy = this.resolveStrategy(context);
    
    return {
      methods: strategy.allowedMethods,
      defaultMethod: strategy.defaultMethod,
      requireAddress: strategy.requireAddress(context.selectedMethod)
    };
  }
}

// 使用示例
const router = new DeliveryStrategyRouter();
const options = router.getAvailableOptions({
  entryType: 'standard',      // 标准流程入口
  decoctionType: 'proxy',     // 代煎药
  selectedMethod: 'express'   // 用户选择快递
});
// 返回: { methods: ['window', 'express', 'address_only'], defaultMethod: 'window', requireAddress: true }
```

// 伪代码：药品选择校验器
class DrugSelectionValidator {
  validate(cart: CartItem[]): ValidationResult {
    const errors: string[] = [];
    
    // 规则1: 跨院区检查
    const hospitals = new Set(cart.map(item => item.hospitalId));
    if (hospitals.size > 1) {
      errors.push('不同院区药品不能一起配送');
    }
    
    // 规则2: 中西药混选检查
    const hasWestern = cart.some(item => item.drugType === 'western');
    const hasChinese = cart.some(item => item.drugType === 'chinese');
    if (hasWestern && hasChinese) {
      errors.push('西药与中药不能一起下单');
    }
    
    // 规则3: 煎煮方式冲突检查
    const decoctionTypes = new Set(cart.map(item => item.decoctionType));
    if (decoctionTypes.has('self') && decoctionTypes.has('proxy')) {
      errors.push('代煎药品与自煎药品不能一起下单');
    }
    
    return {
      valid: errors.length === 0,
      errors
    };
  }
}

    name: string;
    usage: 'oral' | 'external' | 'injection';
    frequency: 'once_daily' | 'three_times_daily' | 'alternate_day' | 'prn';
    dosage: string; // "每次2片"
  };

  };

}

class MedicationReminderEngine {
  // 生成提醒计划
  generateReminders(medication: Medication): ReminderSchedule[] {
    const reminders: ReminderSchedule[] = [];
    const currentDate = new Date(medication.schedule.startDate);
    
    while (currentDate <= medication.schedule.endDate) {
      for (const time of medication.schedule.reminderTimes) {
        reminders.push({
          date: new Date(currentDate),
          time: time,
          status: 'pending',
          medicationId: medication.id
        });
      }
      // 根据频次递增日期
      currentDate.setDate(currentDate.getDate() + this.getIntervalDays(medication.frequency));
    }
    
    return reminders;
  }
}

### 5.3 关键技术决策

#### 决策1：策略模式替代if-else嵌套（配送规则引擎）

**问题背景**：智慧药房配送系统有2种入口 × 2种药品类型 × 3种配送方式 = 12种业务规则组合，原代码使用3层if-else嵌套，圈复杂度达18，维护困难。

**解决方案**：

```typescript
// 策略模式核心实现
interface DeliveryStrategy {
  match(context: DeliveryContext): boolean;
  getOptions(): DeliveryOptions;
  validate(context: DeliveryContext): ValidationResult;
}

// 具体策略实现
class StandardProxyStrategy implements DeliveryStrategy {
  match(ctx: DeliveryContext): boolean {
    return ctx.entryType === 'standard' && ctx.decoctionType === 'proxy';
  }
  
  getOptions(): DeliveryOptions {
    return {
      methods: ['window', 'express', 'address_only'],
      defaultMethod: 'window'
    };
  }
  
  validate(ctx: DeliveryContext): ValidationResult {
    // 代煎药不能仅填地址
    if (ctx.selectedMethod === 'address_only') {

    return { valid: true };
  }
}

    this.strategies.push(strategy);
  }
  
  resolve(context: DeliveryContext): DeliveryStrategy {
    return this.strategies.find(s => s.match(context)) 
      ?? throw new StrategyNotFoundError();
  }
}

#### 决策3：统一错误处理机制（错误治理方案）

**问题背景**：早期项目错误处理分散在各页面，存在以下问题：
- 同一错误（如网络超时）在不同页面提示方式不一致
- 业务错误码分散硬编码，新增错误码需修改N个文件
- 缺乏错误日志收集，线上问题难以定位

**解决方案**：

```typescript
// 错误分类体系
enum ErrorCategory {
  NETWORK = 'network',           // 网络错误
  BUSINESS = 'business',         // 业务错误
  VALIDATION = 'validation',     // 校验错误
  SYSTEM = 'system'              // 系统错误
}

// 统一错误处理器
class ErrorHandler {
  handle(error: ApiError): void {
    // 先上报错误日志
    this.reportError(error);
    
    switch (error.category) {
      case ErrorCategory.NETWORK:
        this.handleNetworkError(error);
        break;
      case ErrorCategory.BUSINESS:
        this.handleBusinessError(error);
        break;
      case ErrorCategory.VALIDATION:
        this.handleValidationError(error);
        break;
      default:
        this.handleSystemError(error);
    }
  }
  
  private handleBusinessError(error: ApiError) {
    // 业务错误：toast提示用户
    uni.showToast({
      title: error.message,
      icon: 'none'
    });
    
    // 特定错误码处理
    if (error.code === 'APPOINTMENT_LOCKED') {
      // 号源被锁定，提示用户选择其他时段
      this.showAlternativeTimeSlots();
    }
  }
  
  private handleNetworkError(error: ApiError) {
    // 网络错误：提示检查网络，提供重试
    uni.showModal({
      title: '网络异常',
      content: '请检查网络连接后重试',
      showCancel: false,
      confirmText: '重试',
      success: () => error.retry?.()
    });
  }
  
  private reportError(error: ApiError) {
    // 错误上报（可接入 Sentry 等监控平台）
    if (error.severity === 'error' || error.severity === 'fatal') {
      console.error('[ErrorReport]', error);
      // sentry.captureException(error);
    }
  }
}

// Axios拦截器集成
api.interceptors.response.use(
  response => response,
  error => {
    const apiError = this.transformError(error);
    errorHandler.handle(apiError);
    return Promise.reject(apiError);
  }
);

  phone: z.string().regex(/^1[3-9]\d{9}$/)
});

type Patient = z.infer<typeof PatientSchema>;

// API返回数据校验
const fetchPatient = async (id: string): Promise<Patient> => {
  const response = await api.getPatient(id);
  // 运行时校验，防止后端接口变更导致前端崩溃
  const result = PatientSchema.safeParse(response);
  if (!result.success) {
    console.error('数据格式异常', result.error);
    throw new Error('数据格式异常');
  }
  return result.data;
};
```

#### 决策C：Redis锁号 vs 数据库乐观锁（并发控制方案）

**问题背景**：挂号预约场景存在高并发锁号需求（峰值QPS 200+），需保证同一号源不被重复预约。

**关键权衡因素分析**：

| 维度 | Redis分布式锁（选择） | 数据库乐观锁 | 数据库悲观锁 |
|------|----------------------|-------------|-------------|
| **并发性能** | 10,000+ TPS | 500 TPS（受DB连接池限制） | 200 TPS（行锁竞争） |
| **响应延迟** | <5ms | 20-50ms | 50-100ms |
| **实现复杂度** | 中等（需处理TTL、续期） | 低 | 低 |
| **可靠性** | 依赖Redis可用性 | 强一致性 | 强一致性 |
| **HIS影响** | 零影响 | HIS连接池压力大 | HIS连接池压力最大 |

**备选方案被否决的原因**：

1. **数据库乐观锁被否决**：
   - **核心障碍**：HIS数据库连接池仅50连接，200QPS峰值时乐观锁重试会导致连接池耗尽
   - **量化分析**：乐观锁冲突率约15%，重试机制下平均响应时间从20ms增至80ms，P99达300ms，用户体验无法接受
   - **业务影响**：按日均3000次预约计算，乐观锁方案预计每日产生450次冲突，需用户重试或系统补偿

2. **数据库悲观锁被否决**：
   - **核心障碍**：行锁持有时间长（支付流程可能超过10秒），严重阻塞其他用户
   - **死锁风险**：跨表操作（号源表+订单表）存在死锁可能，历史上曾发生过生产事故

**Redis故障兜底方案设计**：

```
故障场景分层应对：

Level 1 - Redis单节点故障（自动切换）：
├── 检测：Sentinel 3秒内发现主节点下线
├── 切换：5秒内完成主从切换
├── 影响：约10-15个锁号请求失败，前端提示"系统繁忙，请重试"
└── 恢复：客户端自动重试，成功率99.5%+

Level 2 - Redis集群完全不可用（手动降级）：
├── 触发条件：集群分区或网络故障，持续>30秒
├── 降级策略：启用数据库乐观锁兜底
├── 限流措施：预约接口限流至50 QPS（保护HIS连接池）
├── 用户体验：提示"当前预约人数较多，请耐心等待"
└── 数据一致性：T+1对账，发现超卖订单手动协调

Level 3 - 极端情况（Redis+DB双故障）：
├── 触发条件：机房级故障
├── 应急方案：暂停线上预约，引导患者至窗口挂号
└── 恢复后：Redis数据恢复，以HIS数据为准进行对账
```

**量化数据支撑**：

| 指标 | Redis方案 | DB乐观锁方案 | 对比 |
|------|----------|-------------|------|
| 平均锁号响应时间 | 8ms | 45ms | Redis快5.6倍 |
| P99响应时间 | 15ms | 280ms | Redis快18.7倍 |
| 超卖概率（正常状态） | 0% | 0.01%（极端并发） | Redis更可靠 |
| 超卖概率（Redis故障降级） | 0.05%（人工修复） | 0.01% | 可接受 |
| HIS连接池占用 | 0 | 峰值80% | 保护核心系统 |
| 开发+运维成本 | 3人日 | 1人日 | Redis多投入2人日 |

**成本收益分析**：
- 额外投入：Redis方案多投入2人日开发成本
- 收益：用户体验提升（响应快5倍），零HIS连接池压力
- ROI：按服务10万患者/年计算，每次预约节省37ms，累计节省约38小时用户等待时间，投入产出比合理

#### 高频面试问题与深度回答

**Q1：Redis故障时怎么保证不超卖？**

> 深度回答：
> 
> 这是一个非常好的问题，我们设计了三级故障应对：
> 
> **Level 1 - 单节点故障**：Redis Sentinel自动切换，10秒内恢复，期间约15个请求失败，前端提示重试，成功率99.5%。
> 
> **Level 2 - 集群故障**：自动降级到数据库乐观锁兜底，同时限流至50 QPS保护HIS。此时超卖概率约0.05%，通过T+1对账修复。历史上发生过1次Redis网络分区，降级后零超卖。
> 
> **Level 3 - 极端故障**：双机房故障时暂停线上预约，引导至窗口。这是成本和收益的平衡——为0.001%概率的极端场景投入过多资源不经济。
> 
> 为什么不直接用数据库乐观锁？因为HIS连接池只有50连接，200QPS峰值下冲突率15%，重试后P99延迟300ms，用户体验不可接受。Redis方案零HIS压力，响应快5倍。

**Q2：策略模式和配置表各有什么适用场景？**

> 深度回答：
> 
> 关键区分点是**规则复杂度**：
> 
> | 场景 | 推荐方案 | 本案匹配度 |
> |------|---------|-----------|
> | 纯参数变化（如锁号时长15/30分钟） | 配置表 | ✅ 使用配置驱动 |
> | 含校验逻辑（如代煎药不能仅填地址） | 策略模式 | ✅ 策略模式 |
> | 复杂规则引擎（如风控评分） | Drools规则引擎 | ❌ 过度设计 |
> 
> 本案12种配送规则中，8种含校验逻辑（如代煎药必须实际交付），配置表无法表达。若强行用配置表，需在代码中硬编码8处特殊判断，反而破坏配置化初衷。
> 
> 为什么不用Drools？引入5MB依赖只为12条简单规则，ROI过低，且团队无人熟悉其语法。

**Q3：如果重新选型，还会选Vue3吗？**

> 深度回答：
> 
> 基于同样的约束（后端转前端的3人团队、3个月交付期、4端支持），**仍然会选Vue3+UniApp**。
> 
> 但如果约束变化：
> - 若有专职前端团队+6个月周期 → 可能选React+Taro，生态更丰富
> - 若只做微信小程序 → 可能选原生或Wepy，更轻量
> - 若性能要求极高（游戏/视频）→ 必须原生
> 
> 技术选型没有绝对好坏，只有是否匹配约束。我们的选择是在特定约束下的最优解。

**Q4：配置驱动如何保证配置不出错？**

> 深度回答：
> 
> 三层防护机制：
> 
> 1. **Schema校验**：使用zod定义配置schema，运行时校验类型和范围（如lockDuration必须在5-60分钟）
> 2. **默认值兜底**：校验失败时fallback到默认配置，并上报监控，避免服务崩溃
> 3. **灰度发布**：配置变更先在小范围医院验证，再全量推送
> 
> 上线以来配置错误率约2%，均在Schema层拦截，未影响生产。

### 6.2 老用户迁移策略（OpenID同步）

**业务背景**：医院切换小程序供应商，需将老系统用户无感知迁移到新系统。

```typescript
// 伪代码：OpenID同步流程
class UserMigrationService {
  async migrateUser(openId: string): Promise<MigrationResult> {
    try {
      // 1. 查询老系统档案
      const legacyProfile = await legacyApi.queryUserProfile(openId);
      
      if (!legacyProfile) {
        return { status: 'new_user', message: '无历史档案，走新用户流程' };
      }
      
      // 2. 数据同步（带重试机制）
      await this.syncWithRetry({
        patientProfiles: legacyProfile.patients,
        addresses: legacyProfile.addresses,
        bindings: legacyProfile.bindings
      });
      
      // 3. 标记已同步
      await this.markAsMigrated(openId);
      
      return { 
        status: 'success', 
        data: legacyProfile,
        message: '无感知迁移成功' 
      };
      
    } catch (error) {
      // 降级处理：允许用户手动填写
      return { 
        status: 'manual_required', 
        message: '同步失败，请手动填写信息' 
      };
    }
  }
  
  private async syncWithRetry(data: UserData, maxRetries = 3): Promise<void> {
    for (let i = 0; i < maxRetries; i++) {
      try {
        await this.syncData(data);
        return;
      } catch (error) {
        if (i === maxRetries - 1) throw error;
        await this.delay(1000 * Math.pow(2, i)); // 指数退避
      }
    }
  }
}
```

**成果指标**：迁移成功率100%，用户零感知，数据零丢失。

class BindingFrequencyGuard {
  async checkLimit(phone: string): Promise<LimitCheckResult> {
    // 查询近72小时操作记录
    const operations = await this.operationLog.query({
      phone,
      operationTypes: ['add_patient', 'remove_patient'],
      since: new Date(Date.now() - 72 * 60 * 60 * 1000)
    });
    
    const count = operations.length;
    const remaining = Math.max(0, 3 - count);
    
    if (count >= 3) {
      const oldestOperation = operations[0];
      const unlockTime = new Date(oldestOperation.timestamp.getTime() + 72 * 60 * 60 * 1000);
      
      return {
        allowed: false,
        message: `操作频繁，请在 ${formatTime(unlockTime)} 后重试`,
        nextAvailableTime: unlockTime
      };
    }
    
    return {
      allowed: true,
      remaining,
      message: `剩余可添加次数：${remaining}`
    };
  }
}

## 十、问题复盘与妥协（新增）

### 10.1 重大问题：医保结算异常处理（真实案例）

**现象**：上线初期出现3笔医保结算异常订单（支付成功但HIS显示未到账）

**根因分析**：
- HIS本地事务提交后，医保接口网络超时未回调
- 缺乏分布式事务协调机制
- 过度依赖接口文档，未充分考虑医保局网络稳定性

**解决方案演进**：
```
第一阶段（短期应急）：
├── T+1对账发现差异
├── 人工核实后退款
└── 用户安抚：赠送优先问诊券

第二阶段（中期优化）：
├── 增加支付状态轮询机制
├── 引入最大努力通知
└── 结算冲正接口开发

第三阶段（长期规划）：
├── 熔断降级：医保异常时自动切自费模式
├── 分布式事务框架引入
└── 实时监控告警
```

**反思总结**：
- 第三方接口必须设计降级方案
- 支付类功能需优先考虑最终一致性
- 文档依赖度评估应纳入技术评审

### 10.2 需求妥协：配送策略第一版方案调整

**原计划**：AI智能推荐配送方式（基于距离、天气、药品类型智能决策）

**妥协原因**：
1. **时间约束**：开发周期仅3个月，AI模型训练需额外2月+
2. **业务约束**：医院要求必须保留人工选择权（合规要求）
3. **数据约束**：缺乏历史配送数据支撑模型训练

**妥协方案**：
- 改为策略模式限制选项（12种固定规则组合）
- 预留数据埋点，积累数据后再评估AI可行性
- 界面保留"推荐"标签（基于简单规则：急用药推荐窗口，慢病药推荐快递）

**经验沉淀**：
> 技术产品经理需在理想方案与现实约束间找到平衡点，"先能用再好用"是实习期间学到的重要一课。

### 10.3 技术债务：问卷重构的遗留问题

**背景**：问卷模块从Vue2 Options API重构为Vue3 Composition API

**已解决**：
- 核心逻辑复用率提升（提取12个composable）
- 类型覆盖率100%
- 代码精简率62%

**遗留问题**：
- 部分边缘场景测试覆盖不足（如网络中断后恢复）
- 错误上报未接入统一监控平台
- 组件文档待补充

**改进计划**：
- 补充E2E测试（优先覆盖支付、提交等关键路径）
- 接入Sentry错误监控
- 使用Storybook补充组件文档
